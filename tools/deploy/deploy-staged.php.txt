// =====================================================================
// Deploy sicuro di tema e/o plugin da ZIP caricati in wp-content/uploads/ecm-deploy/
// (ecm.zip per il tema, ecm-core.zip per il plugin, ecm-import-delta.zip facoltativo per il bundle).
// Incollare come "code" di novamira/execute-php (senza tag <?php). Non usa exit/die.
//
// 1. estrae ogni ZIP in una cartella temporanea e controlla i percorsi (niente backslash, niente "..");
// 2. fa il lint di ogni PHP: se c'è un errore non tocca nulla e la versione online resta com'è;
// 3. aggiorna i file AL LORO POSTO (sovrascrive e poi elimina quelli non più presenti): la cartella del tema
//    non sparisce mai, nemmeno per un istante. (Il 20/11/2026 un deploy fatto rinominando la cartella ha fatto
//    attivare a un plugin Hostinger il suo tema "hostinger-ai-theme": sito bianco.)
// 4. controlla che il tema attivo sia ancora "ecm" (e lo riattiva se serve), resetta OPcache,
//    svuota LiteSpeed e la CDN Hostinger (stessa azione del pulsante "Svuota cache" del plugin Hostinger) oppure,
//    sul sito definitivo, la CDN Rocket.net tramite il mu-plugin cdn-cache-management.
// Verificare poi le pagine in una NUOVA chiamata (il codice nuovo vale dalla richiesta successiva).
// =====================================================================
$rm = function ( $dir ) use ( &$rm ) {
	if ( ! is_dir( $dir ) ) { return; }
	$it = new RecursiveIteratorIterator( new RecursiveDirectoryIterator( $dir, FilesystemIterator::SKIP_DOTS ), RecursiveIteratorIterator::CHILD_FIRST );
	foreach ( $it as $f ) { $f->isDir() ? rmdir( $f->getPathname() ) : unlink( $f->getPathname() ); }
	rmdir( $dir );
};
$deploy = function ( $type, $slug ) use ( $rm ) {
	$zip_path = WP_CONTENT_DIR . '/uploads/ecm-deploy/' . $slug . '.zip';
	$base     = 'plugin' === $type ? WP_PLUGIN_DIR : get_theme_root();
	$target   = $base . '/' . $slug;
	$stage    = WP_CONTENT_DIR . '/uploads/ecm-deploy/stage-' . $slug;
	$r        = array( 'slug' => $slug );
	if ( ! file_exists( $zip_path ) ) { return $r + array( 'skipped' => 'zip non caricato' ); }
	$rm( $stage );
	$zip = new ZipArchive();
	if ( true !== $zip->open( $zip_path ) ) { return $r + array( 'error' => 'zip non apribile' ); }
	for ( $i = 0; $i < $zip->numFiles; $i++ ) {
		$n = (string) $zip->getNameIndex( $i );
		if ( false !== strpos( $n, '\\' ) || false !== strpos( $n, '..' ) || 0 !== strpos( $n, $slug . '/' ) ) { $zip->close(); return $r + array( 'error' => 'percorso non valido ' . $n ); }
	}
	wp_mkdir_p( $stage );
	$ok = $zip->extractTo( $stage );
	$zip->close();
	if ( ! $ok ) { return $r + array( 'error' => 'estrazione' ); }
	$src = $stage . '/' . $slug;
	$errors = array(); $count = 0;
	foreach ( new RecursiveIteratorIterator( new RecursiveDirectoryIterator( $src, FilesystemIterator::SKIP_DOTS ) ) as $f ) {
		if ( '.php' !== substr( $f->getFilename(), -4 ) ) { continue; }
		++$count;
		try { token_get_all( (string) file_get_contents( $f->getPathname() ), TOKEN_PARSE ); } catch ( \ParseError $e ) { $errors[] = str_replace( $src, '', $f->getPathname() ) . ':' . $e->getLine() . ' ' . $e->getMessage(); }
	}
	$r['php_files']   = $count;
	$r['lint_errors'] = $errors;
	if ( $errors ) { $rm( $stage ); return $r; }
	// Copia al posto: prima i file nuovi/aggiornati, poi via quelli che non esistono più.
	$keep = array(); $copied = 0;
	foreach ( new RecursiveIteratorIterator( new RecursiveDirectoryIterator( $src, FilesystemIterator::SKIP_DOTS ), RecursiveIteratorIterator::SELF_FIRST ) as $f ) {
		$rel = substr( $f->getPathname(), strlen( $src ) );
		$keep[ $rel ] = true;
		$dst = $target . $rel;
		if ( $f->isDir() ) { wp_mkdir_p( $dst ); continue; }
		if ( ! file_exists( $dst ) || md5_file( $dst ) !== md5_file( $f->getPathname() ) ) { copy( $f->getPathname(), $dst ); ++$copied; }
	}
	$deleted = 0;
	foreach ( new RecursiveIteratorIterator( new RecursiveDirectoryIterator( $target, FilesystemIterator::SKIP_DOTS ), RecursiveIteratorIterator::CHILD_FIRST ) as $f ) {
		$rel = substr( $f->getPathname(), strlen( $target ) );
		if ( isset( $keep[ $rel ] ) ) { continue; }
		$f->isDir() ? @rmdir( $f->getPathname() ) : unlink( $f->getPathname() );
		++$deleted;
	}
	$rm( $stage );
	unlink( $zip_path );
	$r['copied']  = $copied;
	$r['deleted'] = $deleted;
	return $r;
};
$out = array(
	'plugin' => $deploy( 'plugin', 'ecm-core' ),
	'theme'  => $deploy( 'theme', 'ecm' ),
);
$dz = WP_CONTENT_DIR . '/uploads/ecm-deploy/ecm-import-delta.zip';
if ( file_exists( $dz ) ) {
	$z = new ZipArchive();
	if ( true === $z->open( $dz ) ) { wp_mkdir_p( WP_CONTENT_DIR . '/uploads/ecm-import' ); $out['delta'] = $z->extractTo( WP_CONTENT_DIR . '/uploads/ecm-import' ); $z->close(); unlink( $dz ); }
}
if ( 'ecm' !== get_option( 'stylesheet' ) && wp_get_theme( 'ecm' )->exists() ) {
	$out['theme_was'] = get_option( 'stylesheet' );
	switch_theme( 'ecm' );
}
$out['active_theme'] = get_option( 'stylesheet' );
if ( function_exists( 'opcache_reset' ) ) { $out['opcache'] = opcache_reset(); }
do_action( 'litespeed_purge_all' );
if ( defined( 'HOSTINGER_PLUGIN_REST_API_BASE' ) ) {
	do_action( 'rest_api_init', rest_get_server() );
	$res        = rest_do_request( new WP_REST_Request( 'DELETE', '/' . trim( HOSTINGER_PLUGIN_REST_API_BASE, '/' ) . '/clear-cache' ) );
	$out['cdn'] = $res->get_data();
}
// CDN Rocket.net (wvup5sgt7n.wpdns.site): il mu-plugin cdn-cache-management espone purge_everything_cache().
global $wp_filter;
if ( isset( $wp_filter['upgrader_process_complete'] ) ) {
	foreach ( $wp_filter['upgrader_process_complete']->callbacks as $cbs ) {
		foreach ( $cbs as $cb ) {
			if ( is_array( $cb['function'] ) && is_object( $cb['function'][0] ) && method_exists( $cb['function'][0], 'purge_everything_cache' ) ) {
				$cb['function'][0]->purge_everything_cache();
				if ( method_exists( $cb['function'][0], 'flush_deferred_purges' ) ) { $cb['function'][0]->flush_deferred_purges(); }
				$out['cdn_rocket'] = 'svuotata';
			}
		}
	}
}
return $out;
